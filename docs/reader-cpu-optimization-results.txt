===== run.py =====
from pathlib import Path
import subprocess, re, statistics, csv
root = Path(__file__).parent
jobs = [
    ("traversal", "^BenchmarkCPUTraversal$/./^Reflection$/^(Encode|Decode)$"),
    ("structs", "^BenchmarkReflectionStruct$/./^(Encode|Decode)$"),
    ("stream", "^BenchmarkStream(Primitives|Message)$/^(Encode|Decode)$/^Unbuffered$/^Reset$"),
    ("owned", "^BenchmarkAllocationOwnedSmall$/^Bytes8$/^(Binary|String)$"),
    ("skip", "^BenchmarkAllocationSkip$/^(Scalar|Message)$/^Unbuffered$/^Reset$"),
]
values = {}
with (root / "timings.txt").open("w") as out:
    for round_index in range(5):
        order = jobs[round_index:] + jobs[:round_index]
        if round_index % 2:
            order.reverse()
        for job_index, (name, pattern) in enumerate(order):
            versions = ["before", "after"] if (round_index + job_index) % 2 == 0 else ["after", "before"]
            for version in versions:
                args = [str(root / f"{version}.test"), "-test.run", "^$", "-test.bench", pattern, "-test.benchmem", "-test.benchtime=200ms", "-test.count=1"]
                out.write(f"ROUND {round_index + 1} GROUP {name} VERSION {version} COMMAND {args!r}\n")
                result = subprocess.run(args, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, check=True)
                out.write(result.stdout)
                out.flush()
                for line in result.stdout.splitlines():
                    match = re.match(r"(Benchmark\S+)\s+\d+\s+([\d.]+) ns/op.*?\s+(\d+) B/op\s+(\d+) allocs/op", line)
                    if match:
                        benchmark, ns, allocated, count = match.groups()
                        values.setdefault((version, benchmark), []).append((float(ns), int(allocated), int(count)))
            print(f"round {round_index + 1} {name} complete", flush=True)
with (root / "summary.csv").open("w") as out:
    writer = csv.writer(out)
    writer.writerow(["version", "case", "samples", "median_ns", "min_ns", "max_ns", "bytes_min", "bytes_max", "allocs_min", "allocs_max"])
    for (version, benchmark), samples in values.items():
        assert len(samples) == 5, (version, benchmark, len(samples))
        times = [sample[0] for sample in samples]
        allocated = [sample[1] for sample in samples]
        counts = [sample[2] for sample in samples]
        writer.writerow([version, benchmark, len(samples), statistics.median(times), min(times), max(times), min(allocated), max(allocated), min(counts), max(counts)])
print(f"{len(values)} version/case pairs, {sum(len(samples) for samples in values.values())} samples", flush=True)
